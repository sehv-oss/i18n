📄 Custom YAML Loader

This example shows how to create a custom loader that reads .yaml files and uses them as translation sources. 

import type { ILoader } from '@sehv-oss/i18n';
import YAML from 'yaml';

export class YamlLoader implements ILoader {
  readonly extensions = ['.yaml', '.yml'];

  parse(content: string) {
    return YAML.parse(content);
  }
}
